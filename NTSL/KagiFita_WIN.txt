//+------------------------------------------------------------------+
//|  KagiFita_WIN  -  Estrategia de execucao NTSL (Profit)            |
//|                                                                   |
//|  CORRECAO A FAVOR DA FITA                                         |
//|   - Fita VENDENDO  + um Kagi VIRA PARA CIMA  -> vende a mercado   |
//|   - Fita COMPRANDO + um Kagi VIRA PARA BAIXO -> compra a mercado  |
//|  O Kagi mostra o preco andando contra a fita; o robo entra do     |
//|  lado da fita esperando a correcao.                               |
//|                                                                   |
//|  6 Kagi do mesmo ativo, cada um com um tamanho de reversao        |
//|  (K1..K6). Liga/desliga cada um em UsarK1..UsarK6.                |
//|                                                                   |
//|  COMO USAR                                                        |
//|   - Grafico base: WIN em 1 minuto.                                |
//|   - Automacao: "Realizar envio de ordens no fechamento do candle".|
//|   - Quantidade: campo "Quantidade por Ordem" da automacao.        |
//|   - Requer Profit Ultra ou Automacao 2.0 (Asset com periodo Kagi  |
//|     e indicadores de agressao TR).                                |
//|   - Confira se o Kagi de K3 (200) bate com um grafico Kagi 200    |
//|     aberto no Profit. Se nao bater, ajuste o tamanho.             |
//|                                                                   |
//|  v1.00 - primeira versao                                          |
//+------------------------------------------------------------------+
input
  // --- 6 Kagi (ativo, bolsa, periodo Kagi, tamanho da reversao) ---
  K1("WINFUT", feedBMF, itKagiChart, 100);
  K2("WINFUT", feedBMF, itKagiChart, 150);
  K3("WINFUT", feedBMF, itKagiChart, 200);
  K4("WINFUT", feedBMF, itKagiChart, 250);
  K5("WINFUT", feedBMF, itKagiChart, 300);
  K6("WINFUT", feedBMF, itKagiChart, 400);
  UsarK1(true);
  UsarK2(true);
  UsarK3(true);
  UsarK4(true);
  UsarK5(true);
  UsarK6(true);
  KagiMin(1);          // Kagi contra a fita (minimo) para entrar

  // --- Fita ---
  FonteFita(2);        // 1 = PowerMeter (janela em minutos) | 2 = Volume de agressao TR (barras)
  JanelaMin(5);        // FonteFita 1: minutos da fita
  JanelaBarras(5);     // FonteFita 2: barras somadas
  FatorFita(1.2);      // Lado da fita precisa ser X vezes maior que o outro

  // --- Ordem ---
  StopPts(120);        // Stop (pontos)
  AlvoPts(75);         // Alvo (pontos)
  SlipStop(20);        // Limite da ordem stop alem do gatilho (pontos)
  Reverter(true);      // Sinal contrario com posicao aberta: vira a mao

  // --- Horario (HHMM) ---
  HoraInicio(915);
  HoraFimEntradas(1715);
  HoraZerar(1750);

var
  d1, d2, d3, d4, d5, d6         : Integer;  // direcao de cada Kagi: 1 sobe, -1 cai
  nSobe, nCai                    : Integer;
  nViraSobe, nViraCai            : Integer;
  vC, vV                         : Float;    // agressao da barra (FonteFita 2)
  agrC, agrV                     : Float;    // agressao somada da janela
  i                              : Integer;
  fitaCompra, fitaVenda          : Boolean;
  sinalCompra, sinalVenda        : Boolean;
  horarioOk                      : Boolean;

begin
  //================================================================
  //  DIRECAO DOS KAGI
  //  Linha do Kagi subindo: fechamento > abertura. Sem mudanca,
  //  mantem a direcao da barra anterior.
  //================================================================
  if not UsarK1 then d1 := 0
  else if K1.Close > K1.Open then d1 := 1
  else if K1.Close < K1.Open then d1 := -1
  else d1 := d1[1];

  if not UsarK2 then d2 := 0
  else if K2.Close > K2.Open then d2 := 1
  else if K2.Close < K2.Open then d2 := -1
  else d2 := d2[1];

  if not UsarK3 then d3 := 0
  else if K3.Close > K3.Open then d3 := 1
  else if K3.Close < K3.Open then d3 := -1
  else d3 := d3[1];

  if not UsarK4 then d4 := 0
  else if K4.Close > K4.Open then d4 := 1
  else if K4.Close < K4.Open then d4 := -1
  else d4 := d4[1];

  if not UsarK5 then d5 := 0
  else if K5.Close > K5.Open then d5 := 1
  else if K5.Close < K5.Open then d5 := -1
  else d5 := d5[1];

  if not UsarK6 then d6 := 0
  else if K6.Close > K6.Open then d6 := 1
  else if K6.Close < K6.Open then d6 := -1
  else d6 := d6[1];

  // Quantos estao subindo / caindo e quantos viraram nesta barra
  nSobe := 0;
  nCai := 0;
  nViraSobe := 0;
  nViraCai := 0;

  if d1 = 1 then nSobe := nSobe + 1;
  if d2 = 1 then nSobe := nSobe + 1;
  if d3 = 1 then nSobe := nSobe + 1;
  if d4 = 1 then nSobe := nSobe + 1;
  if d5 = 1 then nSobe := nSobe + 1;
  if d6 = 1 then nSobe := nSobe + 1;

  if d1 = -1 then nCai := nCai + 1;
  if d2 = -1 then nCai := nCai + 1;
  if d3 = -1 then nCai := nCai + 1;
  if d4 = -1 then nCai := nCai + 1;
  if d5 = -1 then nCai := nCai + 1;
  if d6 = -1 then nCai := nCai + 1;

  if (d1 = 1) and (d1[1] = -1) then nViraSobe := nViraSobe + 1;
  if (d2 = 1) and (d2[1] = -1) then nViraSobe := nViraSobe + 1;
  if (d3 = 1) and (d3[1] = -1) then nViraSobe := nViraSobe + 1;
  if (d4 = 1) and (d4[1] = -1) then nViraSobe := nViraSobe + 1;
  if (d5 = 1) and (d5[1] = -1) then nViraSobe := nViraSobe + 1;
  if (d6 = 1) and (d6[1] = -1) then nViraSobe := nViraSobe + 1;

  if (d1 = -1) and (d1[1] = 1) then nViraCai := nViraCai + 1;
  if (d2 = -1) and (d2[1] = 1) then nViraCai := nViraCai + 1;
  if (d3 = -1) and (d3[1] = 1) then nViraCai := nViraCai + 1;
  if (d4 = -1) and (d4[1] = 1) then nViraCai := nViraCai + 1;
  if (d5 = -1) and (d5[1] = 1) then nViraCai := nViraCai + 1;
  if (d6 = -1) and (d6[1] = 1) then nViraCai := nViraCai + 1;

  //================================================================
  //  FITA: quem esta agredindo mais
  //================================================================
  agrC := 0;
  agrV := 0;
  if FonteFita = 1 then
  begin
    agrC := PowerMeter(osBuy, JanelaMin);
    agrV := PowerMeter(osSell, JanelaMin);
  end
  else
  begin
    vC := AgressionVolBuy;
    vV := AgressionVolSell;
    for i := 0 to JanelaBarras - 1 do
    begin
      agrC := agrC + vC[i];
      agrV := agrV + vV[i];
    end;
  end;

  fitaCompra := (agrC > 0) and (agrC >= agrV * FatorFita);
  fitaVenda  := (agrV > 0) and (agrV >= agrC * FatorFita);

  //================================================================
  //  SINAIS (correcao contra o Kagi, a favor da fita)
  //================================================================
  sinalVenda  := fitaVenda  and (nViraSobe >= 1) and (nSobe >= KagiMin);
  sinalCompra := fitaCompra and (nViraCai  >= 1) and (nCai  >= KagiMin);

  horarioOk := (Time >= HoraInicio) and (Time < HoraFimEntradas);

  if sinalVenda then PaintBar(clRed)
  else if sinalCompra then PaintBar(clLime);

  //================================================================
  //  ORDENS
  //================================================================
  if HasPosition and (Time >= HoraZerar) then
    ClosePosition
  else if not HasPosition then
  begin
    if horarioOk then
    begin
      if sinalVenda then
        SellShortAtMarket
      else if sinalCompra then
        BuyAtMarket;
    end;
  end
  else if IsBought then
  begin
    if Reverter and horarioOk and sinalVenda then
      ReversePosition
    else
    begin
      SellToCoverLimit(BuyPrice + AlvoPts);
      SellToCoverStop(BuyPrice - StopPts, BuyPrice - StopPts - SlipStop);
    end;
  end
  else if IsSold then
  begin
    if Reverter and horarioOk and sinalCompra then
      ReversePosition
    else
    begin
      BuyToCoverLimit(SellPrice - AlvoPts);
      BuyToCoverStop(SellPrice + StopPts, SellPrice + StopPts + SlipStop);
    end;
  end;
end;
